  const amount =
    Number(verified?.amountPaid);

  if (!Number.isFinite(amount) || amount <= 0) {
    throw new Error(
      "Invalid funding amount"
    );
  }

  console.log(
    "VERIFIED FUNDING AMOUNT:",
    amount
  );

  // -------------------------------------------------------
  // 7. Start the database transaction
  // -------------------------------------------------------

  const client =
    await db.connect();

  try {
    await client.query("BEGIN");

    // -----------------------------------------------------
    // 8. Prevent duplicate funding
    // -----------------------------------------------------

    const existing =
      await client.query(
        `
          SELECT *
          FROM transactions
          WHERE reference = $1
          LIMIT 1
        `,
        [transactionReference]
      );

    if (existing.rows.length > 0) {
      console.log(
        "DUPLICATE MONNIFY TRANSACTION:",
        transactionReference
      );

      await client.query("ROLLBACK");

      return {
        success: true,
        duplicate: true,
        transaction: existing.rows[0],
      };
    }

    // -----------------------------------------------------
    // 9. Lock the wallet before updating its balance
    // -----------------------------------------------------

    console.log(
      "LOCKING WALLET:",
      reservedAccount.wallet_id
    );

    const wallet =
      await findWalletForUpdate(
        reservedAccount.wallet_id,
        client
      );

    if (!wallet) {
      throw new Error(
        "Wallet not found"
      );
    }

    // -----------------------------------------------------
    // 10. Record balances before funding
    // -----------------------------------------------------

    const balanceBefore =
      Number(wallet.available_balance || 0);

    const ledgerBalanceBefore =
      Number(wallet.ledger_balance || 0);

    console.log(
      "WALLET BEFORE FUNDING:",
      {
        walletId: wallet.id,
        availableBalance: balanceBefore,
        ledgerBalance: ledgerBalanceBefore,
        fundingAmount: amount,
      }
    );

    // -----------------------------------------------------
    // 11. Credit the wallet
    // -----------------------------------------------------

    console.log(
      "CREDITING WALLET:",
      {
        walletId: wallet.id,
        amount,
      }
    );

    const updatedWallet =
      await creditWallet(
        wallet.id,
        amount,
        client
      );

    if (!updatedWallet) {
      throw new Error(
        "Failed to credit wallet"
      );
    }

    // -----------------------------------------------------
    // 12. Record balances after funding
    // -----------------------------------------------------

    const balanceAfter =
      Number(updatedWallet.available_balance || 0);

    const ledgerBalanceAfter =
      Number(updatedWallet.ledger_balance || 0);

    console.log(
      "WALLET AFTER FUNDING:",
      {
        walletId: updatedWallet.id,
        availableBalance: balanceAfter,
        ledgerBalance: ledgerBalanceAfter,
      }
    );

    // -----------------------------------------------------
    // 13. Record the transaction
    // -----------------------------------------------------

    console.log(
      "RECORDING FUNDING TRANSACTION:",
      transactionReference
    );

    const transaction =
      await recordFunding(
        {
          walletId: wallet.id,

          reference:
            transactionReference,

            transactionReference,

            paymentReference:
              eventData.paymentReference ||
              null,

            accountNumber,

            amountPaid: amount,

            paymentStatus:
              verified.paymentStatus,

            paymentMethod:
              verified.paymentMethod ||
              eventData.paymentMethod ||
              null,

            currency:
              verified.currency ||
              eventData.currency ||
              "NGN",

            webhook: webhookData,
          },
        },
        client
      );

    // -----------------------------------------------------
    // 14. Commit the wallet credit and transaction
    // -----------------------------------------------------

    await client.query("COMMIT");

    console.log(
      "FUNDING COMMITTED SUCCESSFULLY:",
      {
        transactionReference,
        walletId: wallet.id,
        amount,
        balanceAfter,
      }
    );

    // -----------------------------------------------------
    // 15. Send notification
    //
    // Notification failure must not undo a committed
    // wallet credit.
    // -----------------------------------------------------

    try {
      await notifyWalletFunding({
        userId: reservedAccount.user_id,
        amount,
        balance: balanceAfter,
        reference: transactionReference,
      });

      console.log(
        "FUNDING NOTIFICATION SENT:",
        transactionReference
      );
    } catch (notificationError) {
      console.error(
        "Funding notification failed:",
        notificationError.message
      );
    }

    // -----------------------------------------------------
    // 16. Return the result
    // -----------------------------------------------------

    return {
      success: true,

      transaction,

      wallet: {
        id: updatedWallet.id,
        availableBalance: balanceAfter,
        ledgerBalance: ledgerBalanceAfter,
      },
    };
  } catch (error) {
    // -----------------------------------------------------
    // Roll back unsuccessful database transactions
    // -----------------------------------------------------

    try {
      await client.query("ROLLBACK");
    } catch (rollbackError) {
      console.error(
        "Funding rollback error:",
        rollbackError.message
      );
    }

    console.error(
      "PROCESS FUNDING ERROR:",
      error.message
    );

    throw error;
  } finally {
    // Always release the pooled connection.
    client.release();
  }
}

 * DEBIT WALLET
 * =========================================================
 *
 * Removes money from the available and ledger balances.
 *
 * The available balance check happens inside SQL.

  walletId,
  amount,
  client
) {

    `

      id = $1
      AND available_balance >= $2
    RETURNING *
    `,
    [
      walletId,
      amount,
    ]
  );

  return result.rows[0] || null;
}

/**
 * =========================================================
 * UPDATE TRANSACTION STATUS
 * =========================================================
 *
 * Supported calls:
 *
 * ---------------------------------------------------------
 * 1.
 *
 * updateTransactionStatus(
 *   transactionId,
 *   status
 * )
 *
 * ---------------------------------------------------------
 * 2.
 *
 * updateTransactionStatus(
 *   transactionId,
 *   status,
 *   client
 * )
 *
 * ---------------------------------------------------------
 * 3.
 *
 * updateTransactionStatus(
 *   reference,
 *   status,
 *   metadata,
 *   client
 * )
 *
 * ---------------------------------------------------------
 *
 * The identifier can be either:
 *
 * - transaction UUID
 * - transaction reference
 *
 * Existing metadata is preserved and merged with
 * the new metadata.
 *
 * =========================================================
 */
export async function updateTransactionStatus(
  identifier,
  status,
  metadataOrClient = {},
  possibleClient = db
) {
  let metadata = {};
  let client = possibleClient;


  /**
   * -------------------------------------------------------
   * BACKWARD COMPATIBILITY
   * -------------------------------------------------------
   *
   * If the third argument has a query() function,
   * it is a PostgreSQL client rather than metadata.
   */
  if (
    metadataOrClient &&
    typeof metadataOrClient.query === "function"
  ) {
    client = metadataOrClient;
    metadata = {};
  } else {
    metadata =
      metadataOrClient &&
      typeof metadataOrClient === "object"
        ? metadataOrClient
        : {};

    client = possibleClient || db;
  }


  /**
   * -------------------------------------------------------
   * UPDATE TRANSACTION
   * -------------------------------------------------------
   *
   * We support both transaction UUID and reference.
   *
   * Existing JSONB metadata is merged with the new
   * metadata supplied by the caller.
   */
  const result = await client.query(
    `
    UPDATE transactions
    SET
      status = $2,

      metadata =
        COALESCE(metadata, '{}'::jsonb)
        ||
        $3::jsonb,

      updated_at = CURRENT_TIMESTAMP

    WHERE
      id::text = $1
      OR reference = $1

    RETURNING *
    `,
    [
      String(identifier),
      status,
      JSON.stringify(metadata),
    ]
  );


  /**
   * -------------------------------------------------------
   * RETURN UPDATED TRANSACTION
   * -------------------------------------------------------
   */
  return result.rows[0] || null;
}
