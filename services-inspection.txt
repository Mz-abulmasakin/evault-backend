function parseDetails(body = {}) {
  let details = body.details ?? body;

  if (typeof details === "string") {
    try {
      details = JSON.parse(details);
    } catch {
      throw serviceError("Invalid service request details.");
    }
  }

  if (
    !details ||
    typeof details !== "object" ||
    Array.isArray(details)
  ) {
    throw serviceError(
      "Valid service request details are required."
    );
  }

  let personnel = details.personnel;

  if (typeof personnel === "string") {
    try {
      personnel = JSON.parse(personnel);
    } catch {
      throw serviceError("Invalid personnel information.");
    }
  }

  if (!Array.isArray(personnel)) {
    throw serviceError(
      "Personnel information is required."
    );
  }

  return {
    ...details,
    personnel,
  };
}

// VALIDATE IMAGE CONTENT
// =========================================================

function validateImageFile(file, label) {
  if (
    !file ||
    !Buffer.isBuffer(file.buffer) ||
    file.buffer.length === 0
  ) {
    throw serviceError(`${label} is required.`);
  }

  if (file.buffer.length > 10 * 1024 * 1024) {
    throw serviceError(
      `${label} must not exceed 10 MB.`
    );
  }

  const buffer = file.buffer;

  const isJpeg =
    buffer.length >= 3 &&
    buffer[0] === 0xff &&
    buffer[1] === 0xd8 &&
    buffer[2] === 0xff;

  const isPng =
    buffer.length >= 8 &&
    buffer.subarray(0, 8).equals(
      Buffer.from([
        137, 80, 78, 71,
        13, 10, 26, 10,
      ])
    );

  if (!isJpeg && !isPng) {
    throw serviceError(
      `${label} must be a valid JPEG or PNG image.`
    );
  }

  const actualMime = isJpeg
    ? "image/jpeg"
    : "image/png";

  if (file.mimetype !== actualMime) {
    throw serviceError(
      `${label} file type does not match its actual contents.`
    );
  }

  return actualMime;
}

// =========================================================
// VALIDATE CAC INFORMATION
// =========================================================

function validateCacDetails(details) {
  const entityType = String(details.entityType || "")
    .trim()
    .toLowerCase();

  const supportedTypes = [
    "business",
    "ltd",
    "ngo",
  ];

  if (!supportedTypes.includes(entityType)) {
    throw serviceError("Invalid CAC entity type.");
  }

  const proposedNames = Array.isArray(details.proposedNames)
    ? details.proposedNames
    : [];

  const requiredNames = entityType === "ngo" ? 3 : 2;

  if (
    proposedNames.filter(
      (name) =>
        typeof name === "string" && name.trim()
    ).length < requiredNames
  ) {
    throw serviceError(
      `Please provide at least ${requiredNames} proposed names.`
    );
  }

  if (!String(details.officeAddress || "").trim()) {
    throw serviceError("Office address is required.");
  }

  if (!String(details.businessNature || "").trim()) {
    throw serviceError("Business nature is required.");
  }

  if (entityType === "ltd") {
    const capital = Number(details.shareCapital);

    if (!Number.isFinite(capital) || capital < 1000000) {
      throw serviceError(
        "Limited liability company share capital must be at least ₦1,000,000."
      );
    }
  }

  const minimumPersonnel = entityType === "business" ? 1 : 2;

  if (details.personnel.length < minimumPersonnel) {
    throw serviceError(
      `At least ${minimumPersonnel} personnel record(s) are required.`
    );
  }

  if (details.personnel.length > 10) {
    throw serviceError(
      "A maximum of 10 personnel records is allowed."
    );
  }

  details.personnel.forEach((person, index) => {
    if (!String(person.fullName || "").trim()) {
      throw serviceError(
        `Personnel ${index + 1}: full name is required.`
      );
    }

    if (!/^\d{11}$/.test(String(person.nin || "").trim())) {
      throw serviceError(
        `Personnel ${index + 1}: valid 11-digit NIN is required.`
      );
    }

    if (!String(person.role || "").trim()) {
      throw serviceError(
        `Personnel ${index + 1}: role is required.`
      );
    }
  });

  return entityType;
}

function generateReference() {
  const date = new Date()
    .toISOString()
    .slice(0, 10)
    .replace(/-/g, "");

  const random = randomBytes(4)
    .toString("hex")
    .toUpperCase();

  return `EVS-CAC-${date}-${random}`;
}

// =========================================================
// SUBMIT CAC REQUEST WITH WALLET PAYMENT
// =========================================================

export async function submitCacServiceRequest({
  userId,
  body,
  files = {},
}) {
  if (!userId) {
    throw serviceError(
      "Authenticated user is required.",
      401,
      "USER_ID_REQUIRED"
    );
  }

  const details = parseDetails(body);
  const entityType = validateCacDetails(details);
  const personnel = details.personnel;

  // Calculate fee on the backend.
  const amount = getCacRegistrationFee(entityType);

  // Check all required documents before uploading.
  const documentsToUpload = [];

  personnel.forEach((person, index) => {
    const passportField = `personnel_${index}_passport`;
    const signatureField = `personnel_${index}_signature`;

    const passport = files[passportField]?.[0];
    const signature = files[signatureField]?.[0];

    if (!passport) {
      throw serviceError(
        `Passport photograph is missing for personnel ${index + 1}.`
      );
    }

    if (!signature) {
      throw serviceError(
        `Signature is missing for personnel ${index + 1}.`
      );
    }

    documentsToUpload.push({
      file: passport,
      type: `PERSONNEL_${index + 1}_PASSPORT`,
      label: `Personnel ${index + 1} passport`,
      personIndex: index,
    });

    documentsToUpload.push({
      file: signature,
      type: `PERSONNEL_${index + 1}_SIGNATURE`,
      label: `Personnel ${index + 1} signature`,
      personIndex: index,
    });
  });

  // Validate image contents before database or storage work.
  for (const document of documentsToUpload) {
    document.mimeType = validateImageFile(
      document.file,
      document.label
    );
  }

  const reference = generateReference();
  const uploadedObjects = [];
  const client = await db.connect();

  try {
    await client.query("BEGIN");

    // -----------------------------------------------------
    // LOCK CUSTOMER WALLET
    // -----------------------------------------------------

    const wallet = await findWalletForUpdateByUserId(
      userId,
      client
    );

    if (!wallet) {
      throw serviceError(
        "Wallet not found.",
        404,
        "WALLET_NOT_FOUND"
      );
    }

    if (wallet.status !== "ACTIVE") {
      throw serviceError(
        "Your wallet is not active.",
        403,
        "WALLET_INACTIVE"
      );
    }

    // -----------------------------------------------------
    // CHECK AVAILABLE BALANCE
    // -----------------------------------------------------

    const balanceBefore = Number(
      wallet.available_balance
    );

    if (!Number.isFinite(balanceBefore)) {
      throw serviceError(
        "Unable to verify wallet balance."
      );
    }

    if (balanceBefore < amount) {
      throw serviceError(
        `Insufficient wallet balance. You need ₦${amount.toLocaleString("en-NG")} for this CAC registration.`,
        400,
        "INSUFFICIENT_BALANCE"
      );
    }

    // -----------------------------------------------------
    // CREATE CAC SERVICE REQUEST
    // -----------------------------------------------------

    const request = await createServiceRequest(
      {
        userId,
        reference,
        serviceType: "CAC",
        entityType,

        details: {
          proposedNames: details.proposedNames,
          officeAddress: details.officeAddress,
          businessNature: details.businessNature,
          shareCapital: details.shareCapital ?? null,

          personnel: personnel.map((person) => ({
            fullName: String(person.fullName).trim(),
            nin: String(person.nin).trim(),
            role: String(person.role).trim(),
          })),
        },

        amount,
        currency: "NGN",
        status: "SUBMITTED",
        paymentStatus: "PAID",
      },
      client
    );

    // -----------------------------------------------------
    // UPLOAD PRIVATE PERSONNEL DOCUMENTS
    // -----------------------------------------------------

    for (const document of documentsToUpload) {
      const uploaded = await uploadPrivateDocument({
        buffer: document.file.buffer,
        mimeType: document.mimeType,
        folder: `service-requests/${request.id}`,
      });

      uploadedObjects.push(uploaded.path);

      await createServiceRequestDocument(
        {
          requestId: request.id,
          uploadedByUserId: userId,
          documentType: document.type,

          fileName: safeFileName(
            document.file.originalname
          ),

          mimeType: uploaded.mimeType,
          fileSize: uploaded.size,
          storageBucket: uploaded.bucket,
          storagePath: uploaded.path,
          visibility: "ADMIN",

          metadata: {
            personnelIndex: document.personIndex,
          },
        },
        client
      );
    }

    // -----------------------------------------------------
    // DEBIT CUSTOMER WALLET
    // -----------------------------------------------------

    const updatedWallet = await debitWallet(
      wallet.id,
      amount,
      client
    );

    if (!updatedWallet) {
      throw serviceError(
        "Unable to debit wallet. Please check your balance.",
        400,
        "WALLET_DEBIT_FAILED"
      );
    }

    const balanceAfter = Number(
      updatedWallet.available_balance
    );

    // -----------------------------------------------------
    // CREATE PAYMENT TRANSACTION
    // -----------------------------------------------------

    const paymentTransaction = await createTransaction(
      {
        walletId: wallet.id,
        reference,

        type: TRANSACTION_TYPES.PAYMENT,

        amount,
        balanceBefore,
        balanceAfter,

        status: TRANSACTION_STATUS.SUCCESS,

        description:
          `CAC ${entityType.toUpperCase()} registration fee`,

        metadata: {
          service: "CAC",
          serviceRequestId: request.id,
          serviceReference: reference,
          entityType,
          currency: "NGN",
        },
      },
      client
    );

    // -----------------------------------------------------
    // NOTIFY CUSTOMER
    // -----------------------------------------------------

    await createNotification(
      {
        userId,
        title: "CAC Registration Submitted",

        message:
          `Your CAC ${entityType.toUpperCase()} registration request has been submitted successfully. ₦${amount.toLocaleString("en-NG")} was debited from your wallet.`,

        type: "INFO",

        metadata: {
          serviceType: "CAC",
          requestId: request.id,
          reference,
          amount,
          paymentStatus: "PAID",
          transactionId: paymentTransaction.id,
          status: "SUBMITTED",
        },
      },
      client
    );

    // -----------------------------------------------------
    // COMMIT DATABASE TRANSACTION
    // -----------------------------------------------------

    await client.query("COMMIT");

    return {
      request: {
        id: request.id,
        reference: request.reference,
        serviceType: request.service_type,
        entityType: request.entity_type,
        status: request.status,
        paymentStatus: "PAID",
        amount,
        submittedAt: request.submitted_at,
      },

      transaction: {
        id: paymentTransaction.id,
        reference: paymentTransaction.reference,
        type: paymentTransaction.type,
        amount: Number(paymentTransaction.amount),
        status: paymentTransaction.status,
        balanceBefore,
        balanceAfter,
      },

      message:
        "CAC registration submitted and payment successful.",
    };

  } catch (error) {
    // Roll back request, debit and transaction.
    await client.query("ROLLBACK").catch(() => {});

    // Clean up uploaded documents if the operation fails.
    for (const objectPath of uploadedObjects) {
      try {
        await deletePrivateDocument(objectPath);
      } catch (cleanupError) {
        console.error(
          "CAC DOCUMENT CLEANUP ERROR:",
          cleanupError.message
        );
      }
    }

    throw error;

  } finally {
    client.release();
  }
}

export async function listAdminServiceRequests({
  page = 1,
  limit = 20,
  search = "",
  serviceType = "",
  status = "",
}) {
  const conditions = [];
  const values = [];

  if (search) {
    values.push(`%${search}%`);
    const p = `$${values.length}`;

    conditions.push(`(
      sr.reference ILIKE ${p}
      OR COALESCE(u.full_name, '') ILIKE ${p}
      OR COALESCE(u.email, '') ILIKE ${p}
      OR COALESCE(u.phone_number, '') ILIKE ${p}
    )`);
  }

  if (serviceType) {
    values.push(serviceType);
    conditions.push(`sr.service_type = $${values.length}`);
  }

  if (status) {
    values.push(status);
    conditions.push(`sr.status = $${values.length}`);
  }

  const whereSql = conditions.length
    ? `WHERE ${conditions.join(" AND ")}`
    : "";

  const limitIndex = values.length + 1;
  const offsetIndex = values.length + 2;
  const offset = (page - 1) * limit;

  const [rowsResult, countResult] = await Promise.all([
    db.query(
      `SELECT
         sr.id, sr.user_id, sr.reference,
         sr.service_type, sr.entity_type,
         sr.status, sr.payment_status,
         sr.amount, sr.currency,
         sr.submitted_at, sr.completed_at,
         sr.created_at, sr.updated_at,
         u.full_name AS customer_name,
         u.username AS customer_username,
         u.email AS customer_email,
         u.phone_number AS customer_phone
       FROM service_requests sr
       INNER JOIN users u ON u.id = sr.user_id
       ${whereSql}
       ORDER BY sr.created_at DESC
       LIMIT $${limitIndex}
       OFFSET $${offsetIndex}`,
      [...values, limit, offset]
    ),

    db.query(
      `SELECT COUNT(*)::int AS total
       FROM service_requests sr
       INNER JOIN users u ON u.id = sr.user_id
       ${whereSql}`,
      values
    ),
  ]);

  return {
    rows: rowsResult.rows,
    total: Number(countResult.rows[0]?.total || 0),
  };
}

function makeError(message, statusCode = 400) {
  const error = new Error(message);
  error.statusCode = statusCode;
  return error;

function toPositiveInteger(value, fallback, max) {
  const parsed = Number.parseInt(value, 10);

  if (!Number.isFinite(parsed) || parsed < 1) {
    return fallback;
  }

  return Math.min(parsed, max);
}

function cleanSearch(value) {
  return typeof value === "string"
    ? value.trim().slice(0, 120)
    : "";
}

function pagination(page, limit, total) {
  return {
    page,
    limit,
    total,
    totalPages: Math.ceil(total / limit),
    hasNextPage: page * limit < total,
    hasPreviousPage: page > 1,
  };
}

function safeFileName(name) {
  const cleaned = String(name || "document")
    .replace(/[^a-zA-Z0-9._-]/g, "_")
    .slice(0, 180);

  return cleaned || "document";
}

// =========================================================
// FILE SIGNATURE VALIDATION
// =========================================================

function validateCompletedDocument(file) {
  if (!file?.buffer || !Buffer.isBuffer(file.buffer)) {
    throw makeError("An uploaded document is invalid.", 400);
  }

  if (file.buffer.length > 10 * 1024 * 1024) {
    throw makeError("Each document must not exceed 10 MB.", 400);
  }

  const mimeType = file.mimetype;
  const buffer = file.buffer;

  // JPEG validation
  if (mimeType === "image/jpeg") {
    if (
      buffer.length < 3 ||
      buffer[0] !== 0xff ||
      buffer[1] !== 0xd8 ||
      buffer[2] !== 0xff
    ) {
      throw makeError("An uploaded JPEG file is invalid.", 400);
    }

    return;
  }

  // PNG validation
  if (mimeType === "image/png") {
    const pngSignature = Buffer.from([
      0x89, 0x50, 0x4e, 0x47,
      0x0d, 0x0a, 0x1a, 0x0a,
    ]);

    if (
      buffer.length < 8 ||
      !buffer.subarray(0, 8).equals(pngSignature)
    ) {
      throw makeError("An uploaded PNG file is invalid.", 400);
    }

    return;
  }

  // PDF validation
  if (mimeType === "application/pdf") {
    if (
      buffer.length < 5 ||
      buffer.subarray(0, 5).toString() !== "%PDF-"
    ) {
      throw makeError("An uploaded PDF file is invalid.", 400);
    }

    return;
  }

  throw makeError(
    "Only PDF, JPEG and PNG documents are allowed.",
    400
  );
}

export async function processAdminServiceRequest({
  requestId,
  adminId,
  status,
  adminNotes,
  rejectionReason,
}) {
  if (!requestId) {
    throw makeError("Service request ID is required.", 400);
  }

  if (!adminId) {
    throw makeError("Administrator ID is required.", 400);
  }

  const allowedStatuses = [
    "UNDER_REVIEW",
    "PROCESSING",
    "REJECTED",
  ];

  if (!allowedStatuses.includes(status)) {
    throw makeError("Invalid service processing status.", 400);
  }

  if (
    status === "REJECTED" &&
    (!rejectionReason || !String(rejectionReason).trim())
  ) {
    throw makeError("A rejection reason is required.", 400);
  }

  const updated = await updateAdminServiceRequestStatus({
    requestId,
    adminId,
    status,
    adminNotes: adminNotes
      ? String(adminNotes).trim()
      : null,
    rejectionReason: status === "REJECTED"
      ? String(rejectionReason).trim()
      : null,
  });

  if (!updated) {
    throw makeError("Service request not found.", 404);
  }

  return updated;
}

// =========================================================
// UPLOAD COMPLETED SERVICE DOCUMENTS
// =========================================================

export async function uploadCompletedServiceDocuments({
  requestId,
  adminId,
  files = [],
  adminNotes = null,
}) {
  if (!requestId) {
    throw makeError("Service request ID is required.", 400);
  }

  if (!adminId) {
    throw makeError("Administrator ID is required.", 400);
  }

  if (!Array.isArray(files) || files.length === 0) {
    throw makeError("Please select at least one document.", 400);
  }

  if (files.length > 10) {
    throw makeError("You can upload a maximum of 10 documents.", 400);
  }

  // Validate every document before uploading anything.
  for (const file of files) {
    validateCompletedDocument(file);
  }

  const client = await db.connect();
  const uploadedObjects = [];

  try {
    await client.query("BEGIN");

    // Lock the request to prevent simultaneous completion.
    const requestResult = await client.query(
      `SELECT
         id,
         user_id,
         reference,
         service_type,
         status
       FROM service_requests
       WHERE id::text = $1 OR reference = $1
       LIMIT 1
       FOR UPDATE`,
      [String(requestId)]
    );

    const request = requestResult.rows[0];

    if (!request) {
      throw makeError("Service request not found.", 404);
    }

    if (request.status === "COMPLETED") {
      throw makeError(
        "This service request is already completed.",
        409
      );
    }

    if (request.status === "REJECTED") {
      throw makeError(
        "Rejected requests cannot be completed.",
        409
      );
    }

    const savedDocuments = [];

    // Upload completed documents to private Supabase storage.
    for (const file of files) {
      const uploaded = await uploadPrivateDocument({
        buffer: file.buffer,
        mimeType: file.mimetype,
        folder: `service-requests/${request.id}/completed`,
      });

      uploadedObjects.push(uploaded);

      // Save document metadata in PostgreSQL.
      const document = await createServiceRequestDocument(
        {
          requestId: request.id,
          uploadedByAdminId: adminId,
          documentType: "COMPLETED_DOCUMENT",
          fileName: safeFileName(file.originalname),
          mimeType: file.mimetype,
          fileSize: file.buffer.length,
          storageBucket: uploaded.bucket,
          storagePath: uploaded.path,
          visibility: "CUSTOMER",
          metadata: {
            originalName: file.originalname,
            uploadedAs: "completed_service_document",
          },
        },
        client
      );

      savedDocuments.push(document);
    }

    // Mark the service request as completed.
    const completedResult = await client.query(
      `UPDATE service_requests
       SET
         status = 'COMPLETED',
         assigned_admin_id = $1,
         admin_notes = COALESCE($2, admin_notes),
         completed_at = CURRENT_TIMESTAMP,
         updated_at = CURRENT_TIMESTAMP
       WHERE id = $3
       RETURNING
         id,
         user_id,
         reference,
         service_type,
         entity_type,
         status,
         payment_status,
         completed_at,
         updated_at`,
      [
        adminId,
        adminNotes ? String(adminNotes).trim() : null,
        request.id,
      ]
    );

    const completedRequest = completedResult.rows[0];

    // Notify the customer in the same database transaction.
    await createNotification(
      {
        userId: request.user_id,
        title: "Service Request Completed",
        message: `Your ${request.service_type} service request (${request.reference}) has been completed. Your documents are now available in eVault.`,
        type: "INFO",
        metadata: {
          serviceRequestId: request.id,
          reference: request.reference,
          serviceType: request.service_type,
        },
      },
      client
    );

    await client.query("COMMIT");

    return {
      request: completedRequest,

      documents: savedDocuments.map((document) => {
        const {
          storage_path,
          storage_bucket,
          ...safeDocument
        } = document;

        return safeDocument;
      }),

      message:
        "Documents uploaded and service request completed successfully.",
    };
  } catch (error) {
    try {
      await client.query("ROLLBACK");
    } catch (rollbackError) {
      console.error(
        "Service request rollback failed:",
        rollbackError
      );
    }

    // Remove uploaded files if the database operation failed.
    for (const uploaded of uploadedObjects) {
      try {
        await deletePrivateDocument(uploaded.path);
      } catch (cleanupError) {
        console.error(
          "Failed to clean up uploaded document:",
          cleanupError
        );
      }
    }

    throw error;
  } finally {
    client.release();
  }
}
